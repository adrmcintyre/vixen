<decl> ::= <obj-decl> | <fun-decl>

<obj-decl> ::= obj <ident> { <idents> }

<fun-decl> ::=
    fun <ident> [ <args> ]
        <fun-stmts>
    end

<idents> ::= <ident> | <ident> ',' <idents>

<fun-stmts> ::=
    <fun-stmt>
|
    <fun-stmt>
    <fun-stmts>

<fun-stmt> ::= <stmt> | <return-stmt>

<return-stmt> ::= return | return <expr>

<stmts> ::=
    <stmt>
|
    <stmt>
    <stmts>

<stmt> ::=
    <assign-stmt>
|   <control-stmt>
|   <call-stmt>

<call-stmt> ::= <expr-fun-call>

<lhs-maybe-array> ::= <ident> | <ident> '[' <expr> ']'

<lhs> ::= <lhs-maybe-array> | <lhs> '.' <lhs-maybe-array>

<assign-stmt> ::=
    <ident> = <expr>
|   <ident> '[' <expr> ']' = <expr>
|   <ident>.<ident> = <expr>

<control-stmt> ::= <if-stmt> | <while-stmt> | <repeat-stmt> | <for-stmt>

<if-stmt> ::=
    if <expr>
        <stmts>
    [
    else
        <stmts>
    ]
    endif

<while-stmt> ::=
    while <expr>
        <stmts>
    wend

<repeat-stmt> ::=
    repeat <expr>
        <stmts>
    until <expr>

<for-stmt> ::=
    for <ident> in <expr> to <expr> [ step <expr> ]
        <stmts>
    next

<exprs> ::= <expr> | <expr> ',' <exprs>

<expr> ::= <fun-call> | <literal>

<fun-call> ::=
    <ident> '(' ')'
    <ident> '(' <exprs> ')'

<literal> ::= <integer> | <float> | <string> | <bool>

<integer> ::= [-+]?[0-9]+

<float> ::= [-+]?[0-9]+[.][0-9]*([e][-+]?[0-9]+)

<bool> ::= false | true

<string> ::= "([^"\] | \[g-z] | \[0-9a-f][0-9a-f])*"
